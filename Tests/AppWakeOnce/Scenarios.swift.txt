extension AppDelegate {
    static func testApp(legacyPreference: Bool = true) -> AppDelegate {
        TestDefaults.standard.values = ["wakeOnceEnabled": legacyPreference]
        let app = AppDelegate()
        app.installTestCallbacks()
        app.render()
        return app
    }
    func testAlways(_ on: Bool) { toggle.state = on ? .on : .off; toggleChanged() }
    func testOnce(_ on: Bool) { onceToggle.state = on ? .on : .off; onceChanged() }
    func testRegister(_ error: String? = nil) { registration.completions.removeFirst()(error) }
    func testLid(_ closed: Bool) { lidMonitor.emit(closed) }
    func testLidFailure() { lidMonitor.fail() }
    func testLidSetupFailure() { lidMonitor.failStart = true }
    func testReply(_ index: Int, _ active: Bool, _ error: String = "") { client.requests[index].reply(active, error) }
    func testHeartbeat() { tick() }
    var testRequests: [String] { client.requests.map { $0.action == "set" ? "set:\($0.enabled)" : $0.action } }
    var testEnabled: Bool { enabled }
    var testBusy: Bool { busy }
    var testIdle: Bool { idle.active }
    var testTracking: Bool { lidMonitor.monitoring }
    var testNote: String { testMessage }
    var testModeEnabled: Bool { onlyOnce }
    var testSwitches: String { "\(toggle.state == .on):\(onceToggle.state == .on)" }
}
var checks = 0
func check(_ condition: @autoclosure () -> Bool, _ label: String) {
    guard condition() else { fatalError(label) }
    checks += 1
    print("PASS: \(label)")
}
func running(once: Bool = true) -> AppDelegate {
    let app = AppDelegate.testApp()
    if once { app.testOnce(true) } else { app.testAlways(true) }
    app.testRegister(); app.testReply(0, true)
    return app
}

do {
    let app = AppDelegate.testApp(legacyPreference: true)
    check(!app.testEnabled && !app.testModeEnabled && app.testSwitches == "false:false", "restart ignores legacy persistent preference and starts with both off")
    app.testOnce(true)
    check(app.testBusy, "once switch directly starts activation without a master switch")
    app.testRegister(); app.testReply(0, true)
    check(app.testEnabled && app.testSwitches == "false:true", "once activation renders only once switch on")
    app.testOnce(false)
    check(app.testRequests == ["set:true", "set:false"], "turning once off directly requests restoration")
    app.testReply(1, false)
    check(!app.testEnabled && app.testSwitches == "false:false", "once off leaves both switches off")
}
do {
    let app = running(once: false)
    check(app.testSwitches == "true:false" && !app.testTracking, "always activation renders only always switch on")
    app.testAlways(false); app.testReply(1, false)
    check(app.testSwitches == "false:false" && !app.testEnabled, "always off restores power and leaves both off")
}
do {
    let app = AppDelegate.testApp()
    app.testOnce(true)
    app.testLid(true); app.testLid(false)
    check(app.testBusy && app.testRequests.isEmpty, "registration retains close/open without sending an enable")
    app.testRegister()
    check(!app.testEnabled && !app.testBusy && app.testRequests.isEmpty && app.testSwitches == "false:false", "registration completion honors pending stop")
    app.testLid(true)
    check(app.testRequests.isEmpty, "later lid close never enables automatically")
}
do {
    let app = AppDelegate.testApp()
    app.testOnce(true); app.testRegister()
    app.testLid(true); app.testLid(false)
    check(app.testRequests == ["set:true"] && !app.testIdle, "opening during enable waits and releases idle assertions")
    app.testReply(0, true)
    check(app.testRequests == ["set:true", "set:false"], "enable reply is followed by exactly one disable")
    app.testReply(1, false)
    check(!app.testEnabled && !app.testBusy && app.testNote.contains("已自动关闭") && app.testSwitches == "false:false", "confirmed automatic restoration turns both switches off")
    app.testReply(0, true)
    check(!app.testEnabled && app.testRequests.count == 2 && app.testSwitches == "false:false", "stale enable reply cannot revive a finished cycle")
    app.testLid(true); app.testLid(false)
    check(app.testRequests.count == 2 && !app.testTracking, "finished cycle ignores later lid changes")
    app.testOnce(true); app.testRegister(); app.testReply(2, true)
    check(app.testEnabled && app.testTracking && app.testSwitches == "false:true", "new manual once activation starts a fresh cycle")
}
do {
    let app = running()
    app.testHeartbeat()
    check(app.testRequests == ["set:true", "heartbeat"], "heartbeat is outstanding before opening")
    app.testLid(true); app.testLid(false)
    app.testReply(2, false)
    app.testReply(1, true)
    check(!app.testEnabled && !app.testBusy && app.testSwitches == "false:false", "stale heartbeat after stop cannot re-enable")
}
do {
    let app = running(once: false)
    app.testOnce(true)
    check(app.testSwitches == "false:true" && app.testTracking && app.testRequests == ["set:true"], "always-to-once changes mode without cycling power or replacing recovery journal")
    app.testLid(true)
    app.testAlways(true)
    check(app.testSwitches == "true:false" && !app.testTracking && app.testRequests == ["set:true"], "once-to-always cancels tracking without restarting backend")
    app.testLid(false)
    check(app.testEnabled && app.testIdle && app.testRequests == ["set:true"], "always mode ignores opening after mode switch")
    app.testOnce(true)
    app.testLid(true); app.testLid(false)
    check(app.testRequests == ["set:true", "set:false"], "switching back to once starts a fresh closing cycle")
}
do {
    let app = running(once: false)
    app.testLidSetupFailure()
    app.testOnce(true)
    check(app.testEnabled && app.testIdle && app.testSwitches == "true:false" && !app.testModeEnabled, "failed once sensor setup retains existing always session")
    check(app.testRequests == ["set:true"] && app.testNote.contains("setup failed"), "sensor setup failure does not mutate backend")
}
do {
    let app = AppDelegate.testApp()
    app.testLidSetupFailure(); app.testOnce(true)
    check(!app.testEnabled && app.testSwitches == "false:false" && app.testRequests.isEmpty, "unavailable sensor blocks direct once activation safely")
}
do {
    let app = running()
    app.testLidFailure()
    check(app.testRequests == ["set:true", "set:false"] && !app.testIdle, "monitor failure requests restoration")
    app.testReply(1, false, "TEST: recovery failed")
    check(!app.testEnabled && app.testNote == "TEST: recovery failed" && app.testSwitches == "false:false", "restoration error does not report success")
}
do {
    let app = running()
    app.testLid(true); app.testLid(false)
    app.testReply(1, true, "TEST: still active")
    check(!app.testEnabled && app.testNote.contains("尚未确认") && app.testSwitches == "false:false", "unexpected active stop reply is shown as unconfirmed")
    app.testHeartbeat()
    check(app.testRequests.count == 2, "unconfirmed stop does not keep session alive with heartbeats")
}
do {
    let app = AppDelegate.testApp()
    app.testOnce(true); app.testLidFailure(); app.testRegister()
    check(app.testRequests.isEmpty && !app.testEnabled && app.testSwitches == "false:false", "monitor failure during registration prevents enable")
}
print("\(checks) AppDelegate wake-once integration checks passed; no system power changes")
