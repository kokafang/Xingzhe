final class FakeServiceClient {
    struct Request {
        let action: String
        let enabled: Bool
        let reply: (Bool, String) -> Void
    }
    var onDisconnect: (() -> Void)?
    var requests: [Request] = []
    var disconnects = 0
    func request(_ action: String, enabled: Bool = false, completion: @escaping (Bool, String) -> Void) {
        requests.append(Request(action: action, enabled: enabled, reply: completion))
    }
    func disconnect() { disconnects += 1 }
}

final class FakeLidMonitor {
    var onChange: ((Bool) -> Void)?
    var onFailure: ((String) -> Void)?
    var closed = false
    var monitoring = false
    var failStart = false
    func start() throws -> Bool {
        monitoring = false
        if failStart { throw AwakeError(text: "TEST: lid setup failed") }
        monitoring = true
        return closed
    }
    func stop() { monitoring = false }
    func emit(_ closed: Bool) {
        self.closed = closed
        if monitoring { onChange?(closed) }
    }
    func fail() {
        guard monitoring else { return }
        monitoring = false
        onFailure?("TEST: lid unavailable")
    }
}
